The following uses only the installed R package. Rendering this document is
optional; the same code runs in an ordinary R session.

```{r}
library(plotfoundry)
s <- pf_theme("tufte", mode="light", profile="reading")
d <- data.frame(x=rep(1:3,3), y=c(1,2,1,3,4,3,5,6,5),
                category=rep(c("A","B","C"),each=3))
p <- ggplot2::ggplot()+pf_line(d,"x","y","category",s,c("A","B","C"))+
  pf_theme_gg(s)+ggplot2::labs(title="Observations",x="Time",y="Value")
pf_export(s,"observations",plot=p)
```
